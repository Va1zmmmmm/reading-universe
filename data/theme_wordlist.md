## 本表如何重生成（2026-09-30 补）

import json, collections
d = json.load(open('data/theme_tags.json', encoding='utf-8'))
c = collections.Counter(t for v in d.values() for t in (v.get('tags') or []))
print(len(d), sum(1 for v in d.values() if v.get('tags')), len(c))
